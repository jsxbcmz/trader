VAR1A:=(HHV(HIGH,4)-CLOSE)/(HHV(HIGH,4)-LLV(LOW,4))*100-90;
VAR2A:=SMA(VAR1A,4,1)+100;
VAR3A:=(CLOSE-LLV(LOW,4))/(HHV(HIGH,4)-LLV(LOW,4))*100;
VAR4A:=SMA(VAR3A,6,1);
VAR5A:=SMA(VAR4A,6,1)+100;
VAR6A:=VAR5A-VAR2A;
砖型图:=IF(VAR6A>4,VAR6A-4,0),COLORRED;

{基础变量定义}
当日收盘价:=C;
短趋线:=EMA(EMA(当日收盘价,10),10);
多空线:=(MA(当日收盘价,14) + MA(当日收盘价,28) +MA(当日收盘价,57) + MA(当日收盘价,114))/4;

{绿转红核心前提}
AA:= REF(砖型图,1) < 砖型图;{当日是红柱}
绿转红:=REF(AA,1) = 0 AND AA = 1;{前一天非红柱，当天红柱}

{有效红柱/绿柱}
有效红柱:=砖型图 > 0;
当天红柱高:=砖型图 - REF(砖型图,1);
有效绿柱:=REF(砖型图,2) - REF(砖型图,1) AND (REF(砖型图,2) - REF(砖型图,1)) > 0;{前一天绿柱高度>0}
前绿柱高:=REF(砖型图,2) - REF(砖型图,1);

{红柱>=绿柱 70%}
绿柱70%:=前绿柱高 * 0.7;
长度条件:=当天红柱高 > 绿柱70%;

{股价区间约束}
短趋线上限:= 短趋线 * 1.02;
{连续两天站多空线}
当日站线:=当日收盘价 > 多空线;
前日站线:=REF(当日收盘价,1) > REF(多空线,1);
连两日站线:=当日站线 AND 前日站线;

股价区间:=连两日站线 AND 当日收盘价 < 短趋线上限;
趋势条件:=短趋线 > 多空线;

价格条件:=当日收盘价 < 60;

选股:绿转红 AND 有效红柱 AND 有效绿柱 AND 长度条件 AND 股价区间  AND 趋势条件 AND 价格条件;
