# 单针下20 副图指标开发方案

## 1. 需求背景

将通达信风格指标"单针下20"集成到四联图的副图指标选项中，用户可通过"副图指标 ▼"下拉框勾选启用，在副图面板中展示三条周期曲线。

**默认不启用**，保持现有三副图（成交额/砖型差值/KDJ）布局不变。

## 2. 指标公式

```
短期: 100*(C-LLV(L,3))/(HHV(C,3)-LLV(L,3))   — 白色 COLORWHITE
中期: 100*(C-LLV(L,14))/(HHV(C,14)-LLV(L,14)) — 黄色 COLORYELLOW
长期: 100*(C-LLV(L,28))/(HHV(C,28)-LLV(L,28)) — 品红 COLORMAGENTA
```

| 参数 | 说明 |
|------|------|
| `LLV(L,N)` | N日最低价的滚动最小值 |
| `HHV(C,N)` | N日收盘价的滚动最大值 |
| 值域 | 0~100 |
| 特征 | 与KDJ面板同构：三条曲线 + 20/80参考线 |

## 3. 技术方案

### 3.1 整体架构

单针下20 在架构上与 KDJ 面板完全同构，遵循现有的副图指标扩展模式：

```
SubChartType.NEEDLE20 (chart_layout.py)
    │
    ├── compute_needle20_indicator()     → chart_indicators.py（计算层）
    ├── Needle20Items + create_needle20_items()  → chart_layout.py（UI组件层）
    ├── build_needle20_label_html()      → chart_overlays.py（标签层）
    └── _update_needle20_panel()         → widgets.py（渲染集成层）

```

## 4. 详细设计

### 4.1 计算层 — `app/chart_indicators.py`

新增 `compute_needle20_indicator(high, low, close)` 函数，复用已有的 `rolling_min()` 和 `rolling_max()`：

```python
def compute_needle20_indicator(high: np.ndarray, low: np.ndarray, close: np.ndarray):
    """单针下20指标：三周期价格位置百分比。"""
    # 短期(3)
    llv_l3 = rolling_min(low, 3)
    hhv_c3 = rolling_max(close, 3)
    span3 = hhv_c3 - llv_l3
    safe_span3 = np.where(np.abs(span3) < 1e-12, np.nan, span3)
    short = (close - llv_l3) / safe_span3 * 100.0

    # 中期(14)
    llv_l14 = rolling_min(low, 14)
    hhv_c14 = rolling_max(close, 14)
    span14 = hhv_c14 - llv_l14
    safe_span14 = np.where(np.abs(span14) < 1e-12, np.nan, span14)
    mid = (close - llv_l14) / safe_span14 * 100.0

    # 长期(28)
    llv_l28 = rolling_min(low, 28)
    hhv_c28 = rolling_max(close, 28)
    span28 = hhv_c28 - llv_l28
    safe_span28 = np.where(np.abs(span28) < 1e-12, np.nan, span28)
    long = (close - llv_l28) / safe_span28 * 100.0

    return {"short": short, "mid": mid, "long": long}
```

```python
def create_needle20_items(needle20_plot) -> Needle20Items:
    short_curve = needle20_plot.plot(pen=pg.mkPen((255, 255, 255), width=1.5), name="短期")
    mid_curve = needle20_plot.plot(pen=pg.mkPen((255, 215, 0), width=1.5), name="中期")
    long_curve = needle20_plot.plot(pen=pg.mkPen((255, 0, 255), width=1.5), name="长期")

    low_line = pg.InfiniteLine(angle=0, movable=False,
        pen=pg.mkPen((0, 176, 80), width=1, style=QtCore.Qt.DotLine))
    needle20_plot.addItem(low_line, ignoreBounds=True)
    low_line.setPos(20)

    high_line = pg.InfiniteLine(angle=0, movable=False,
        pen=pg.mkPen((255, 0, 0), width=1, style=QtCore.Qt.DotLine))
    needle20_plot.addItem(high_line, ignoreBounds=True)
    high_line.setPos(80)

    v_line = pg.InfiniteLine(angle=90, movable=False,
        pen=pg.mkPen((200, 200, 200), width=1))
    needle20_plot.addItem(v_line, ignoreBounds=True)
    v_line.hide()

    label = QtWidgets.QLabel(needle20_plot)
    label.setStyleSheet("color: white; background: transparent; border: none; padding: 0px; margin: 0px;")
    label.setTextFormat(QtCore.Qt.RichText)
    label.move(40, 2)
    label.hide()

    return Needle20Items(
        short_curve=short_curve, mid_curve=mid_curve, long_curve=long_curve,
        low_line=low_line, high_line=high_line, v_line=v_line, label=label,
    )
```

#### 4.4.3 新增渲染方法

```python
def _update_needle20_panel(self, x, h, l, c):
    self.needle20Plot.clear()
    self.needle20Plot.addItem(self.needle20ShortCurve)
    self.needle20Plot.addItem(self.needle20MidCurve)
    self.needle20Plot.addItem(self.needle20LongCurve)
    self.needle20Plot.addItem(self.needle20LowLine, ignoreBounds=True)
    self.needle20LowLine.setPos(20)
    self.needle20Plot.addItem(self.needle20HighLine, ignoreBounds=True)
    self.needle20HighLine.setPos(80)
    self.needle20Plot.addItem(self.needle20VLine, ignoreBounds=True)
    self.needle20VLine.show()

    result = compute_needle20_indicator(h, l, c)
    self._needle20_short_values = result["short"]
    self._needle20_mid_values = result["mid"]
    self._needle20_long_values = result["long"]
    self.needle20ShortCurve.setData(x, self._needle20_short_values)
    self.needle20MidCurve.setData(x, self._needle20_mid_values)
    self.needle20LongCurve.setData(x, self._needle20_long_values)

    # Y轴范围：始终包含 [0, 100]
    all_vals = np.concatenate([
        self._needle20_short_values[np.isfinite(self._needle20_short_values)],
        self._needle20_mid_values[np.isfinite(self._needle20_mid_values)],
        self._needle20_long_values[np.isfinite(self._needle20_long_values)],
    ])
    if len(all_vals) > 0:
        ymin = min(float(np.min(all_vals)), 0.0)
        ymax = max(float(np.max(all_vals)), 100.0)
        pad = max((ymax - ymin) * 0.06, 1.0)
        self.needle20Plot.setYRange(ymin - pad, ymax + pad, padding=0)
    else:
        self.needle20Plot.setYRange(-5, 105, padding=0)
```

```python
def _update_needle20_label_values(self, idx: int):
    if (idx < 0 or idx >= len(self._needle20_short_values) or
        idx >= len(self._needle20_mid_values) or idx >= len(self._needle20_long_values)):
        self.needle20Label.hide()
        return

    text = build_needle20_label_html(
        float(self._needle20_short_values[idx]),
        float(self._needle20_mid_values[idx]),
        float(self._needle20_long_values[idx]),
    )
    self.needle20Label.setText(text)
    self.needle20Label.adjustSize()
    self.needle20Label.show()
```
