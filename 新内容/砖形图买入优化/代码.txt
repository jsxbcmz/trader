{个股主力操盘习惯不一，有的票更适合买入战法，萝卜青菜各有所爱；

增加最大回撤幅度、连续做错次数及复利}
命运上升:=SLOPE(命值,7)>-0.02 AND 运值>命值;
黄线绿转红:=CC>0 AND C>命值 AND 命运上升;
强红:=ABS(砖型图-REF(砖型图,1))>2/3*ABS(REF(砖型图,1)-REF(砖型图,2));
买入:=黄线绿转红 AND 强红;
STICKLINE(买入,砖型图,REF(砖型图,1),3,0),COLOR008CFF;

{复利信息}
五日涨幅:=MAX(HHV(REFX(C,5),5)/C-1,-100);
次日涨幅:=IF(CURRBARSCOUNT=1,C,REFX(C,1))/C-1;
买入次数:=SUM(买入,0);
脱成本:=IF(次日涨幅>=0.05,1,0);
总涨幅:=IF(脱成本,五日涨幅,次日涨幅);

最大回撤:LLV(总涨幅,0),NODRAW;
连续做错次数:HHV(COUNT(买入 AND 0>总涨幅,BARSLAST(买入 AND 总涨幅>0)),0),NODRAW;
胜次:=SUM(IF(买入 AND 总涨幅>0.02,1,0),0);
均值:=SUM(IF(买入 AND 总涨幅!=-100,总涨幅,0),0)/MAX(买入次数,1);
复利:=MULAR(IF(买入 AND 总涨幅!=-100,1+总涨幅,1),0);
胜率:MAX(胜次,0)/MAX(买入次数,1),NODRAW;
总天数:=BARSCOUNT(买入);

DRAWTEXT_FIX(ISLASTBAR,0.1,0.13,0,'战绩:'),COLORYELLOW;
DRAWNUMBER_FIX(ISLASTBAR,0.2,0.13,0,总天数),COLORRED;
DRAWNUMBER_FIX(ISLASTBAR,0.38,0.13,0,均值*100),COLORRED;
DRAWNUMBER_FIX(ISLASTBAR,0.55,0.13,0,复利*100),COLORRED;
