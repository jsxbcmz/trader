{跳空}    
{剔除跳空选股迭代优化点：
跳空后及时回补则不认为是跳空

自定义最近跳空查询的周期长度（暂不支持历史任意天回测）

自定义跳空幅度避免错过强势票（航天发展）}
周期:=40;
跳空幅度:=0.0;
跳空0:=(L>REF(H,1)*(1+跳空幅度) AND 周期>CURRBARSCOUNT);
跳空值0:=VALUEWHEN(跳空0=1,REF(H,1));
跳空值1:=VALUEWHEN(跳空0=1 AND ((COUNT(REF(跳空0=1,1),-1)>0 AND REF(跳空值0,1)>=LLV(REF(L,1)*(1-跳空幅度),MAX(BARSLAST(REF(跳空0,1)),1))) OR COUNT(REF(跳空0=1,1),-1)=0),跳空值0);
跳空:=IF(跳空0=1,1-(跳空值1!=跳空值0 AND COUNT(跳空0,-1)>1),0);
跳空值:=IF(COUNT(跳空0,-1)=1,跳空值0,跳空值1);
回补:(跳空值>=LLV(L,MAX(BARSLAST(跳空),1))*(1-跳空幅度)) OR COUNT(跳空=1,-1)=0;
